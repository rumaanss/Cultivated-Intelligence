cmake_minimum_required(VERSION 3.20)
project(ci_plants LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_library(ci_plants "${ROOT}/src/plants/plant_world.cpp")
target_include_directories(ci_plants PUBLIC "${ROOT}/include")
add_executable(ci_plant_sim "${ROOT}/src/plants/plant_main.cpp")
target_link_libraries(ci_plant_sim PRIVATE ci_plants)
add_executable(test_plants "${ROOT}/tests/plants/test_plants.cpp")
target_link_libraries(test_plants PRIVATE ci_plants)
enable_testing()
add_test(NAME Plants COMMAND test_plants)
if(MSVC)
 target_compile_options(ci_plants PRIVATE /W4)
else()
 target_compile_options(ci_plants PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(test_reproduction_intent "${ROOT}/tests/plants/test_reproduction_intent.cpp" "${ROOT}/src/brain/rnn_brain.cpp" "${ROOT}/src/world/ecology.cpp" "${ROOT}/src/world/celestial.cpp" "${ROOT}/src/world/chemistry.cpp")
target_include_directories(test_reproduction_intent PRIVATE "${ROOT}/include")
add_test(NAME ReproductionIntent COMMAND test_reproduction_intent)
add_executable(plant_showcase "${ROOT}/tools/plants/showcase.cpp")
target_link_libraries(plant_showcase PRIVATE ci_plants)
