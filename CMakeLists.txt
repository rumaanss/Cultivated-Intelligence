# Optional MSVC PGO applies to the maintained native applications and libraries.
set(CYBERNET_PGO_MODE "OFF" CACHE STRING "OFF, INSTRUMENT, or OPTIMIZE (MSVC only)")
set_property(CACHE CYBERNET_PGO_MODE PROPERTY STRINGS OFF INSTRUMENT OPTIMIZE)
if(NOT CYBERNET_PGO_MODE MATCHES "^(OFF|INSTRUMENT|OPTIMIZE)$")
 message(FATAL_ERROR "Invalid CYBERNET_PGO_MODE")
endif()
if(NOT CYBERNET_PGO_MODE STREQUAL "OFF")
 if(NOT MSVC)
  message(FATAL_ERROR "PGO modes require MSVC")
 endif()
 foreach(target ci_core ci_transport ci_headless ci_research ci_aquatic_core ci_aquatic)
  if(TARGET ${target})
   target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:/GL>")
   if(target MATCHES "^ci_(headless|research|aquatic)$")
    target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/INCREMENTAL:NO>")
    if(CYBERNET_PGO_MODE STREQUAL "INSTRUMENT")
     target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/LTCG:PGINSTRUMENT>")
    else()
     target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:/LTCG:PGOPTIMIZE>")
    endif()
   endif()
  endif()
 endforeach()
endif()

if(CI_BUILD_TESTS)
 foreach(testDirectory simulation simulation/tools/plants)
  get_property(testTargets DIRECTORY ${testDirectory} PROPERTY BUILDSYSTEM_TARGETS)
  foreach(testTarget IN LISTS testTargets)
   if(testTarget MATCHES "^test_")
    if(MSVC)
     target_compile_options(${testTarget} PRIVATE /UNDEBUG)
    else()
     target_compile_options(${testTarget} PRIVATE -UNDEBUG)
    endif()
   endif()
  endforeach()
 endforeach()
endif()
